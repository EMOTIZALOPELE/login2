#include <WiFi.h>
#include <WiFiClient.h>
#include <HTTPClient.h>
#include <WebServer.h>
#include <ESP32Servo.h>
#include <DNSServer.h>
#include <Preferences.h>
#include <ESPmDNS.h> // <-- 1. LIBRERÍA AÑADIDA

// --- Parámetros para el Modo de Configuración (AP) ---
const char* AP_SSID = "ESP32_Config_Servo";
const char* AP_PASSWORD = "password";
IPAddress AP_LOCAL_IP(192, 168, 4, 1);
IPAddress AP_GATEWAY(192, 168, 4, 1);
IPAddress AP_SUBNET(255, 255, 255, 0);
WebServer server(80);
DNSServer dnsServer;

// --- Parámetros para el Modo de Operación (STA) ---
String STA_ssid = "";
String STA_password = "";
// Ya no necesitamos la IP del servidor aquí para esta lógica, pero la dejamos por si se usa en otro lado
const char* serverName = "http://192.168.0.170/login2/public/index.php/funcional/actualizarEstado";

// --- Parámetros y variables del Servo ---
Servo servo;
bool isOpened = false;
uint8_t servoPin = 2;

// --- Otros ---
Preferences preferences;
bool wifiConfigured = false;

// --- Funciones del Modo de Operación (Control del Servo y Comunicación HTTP) ---
void enviarEstado(String estado) {
  // Esta función no necesita cambios, seguirá reportando al servidor de tu PC
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String url = String(serverName) + "/" + estado;
    http.begin(url);
    int httpResponseCode = http.GET();
    if (httpResponseCode > 0) {
      Serial.print("Respuesta del servidor: ");
      Serial.println(http.getString());
    } else {
      Serial.print("Error en la solicitud HTTP: ");
      Serial.println(httpResponseCode);
    }
    http.end();
  } else {
    Serial.println("No hay conexión Wi-Fi, no se puede enviar el estado.");
  }
}

void handle_servoOpen() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  isOpened = true;
  servo.write(180);
  // La lógica de 'enviarEstado' se ha movido para no depender de ella para la respuesta
  server.send(200, "text/plain", "Servo abierto");
  enviarEstado("abierto"); // Se envía después para no retrasar la respuesta al navegador
}

void handle_servoClose() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  isOpened = false;
  servo.write(0);
  server.send(200, "text/plain", "Servo cerrado");
  enviarEstado("cerrado");
}

void handle_OnConnect() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  String html = "<html><body>";
  html += "<h1>Control del Servo</h1>";
  html += "<p>Estado actual: <b>" + String(isOpened ? "ABIERTO" : "CERRADO") + "</b></p>";
  html += "<p><a href='/servoOpen'><button>Abrir Servo</button></a></p>";
  html += "<p><a href='/servoClose'><button>Cerrar Servo</button></a></p>";
  html += "</body></html>";
  server.send(200, "text/html", html);
}

void handle_NotFound() {
  if (wifiConfigured) {
    server.send(404, "text/plain", "Página no encontrada");
  } else {
    server.send(200, "text/html", "Redireccionando a la pagina de configuracion...");
  }
}

// --- Funciones del Modo de Configuración (AP) ---
// (Las funciones startAPMode, handleAPRoot y handleAPSave no necesitan cambios)
void startAPMode() {
  Serial.println("Iniciando modo de configuración (AP)...");
  WiFi.softAPConfig(AP_LOCAL_IP, AP_GATEWAY, AP_SUBNET);
  WiFi.softAP(AP_SSID, AP_PASSWORD);
  Serial.print("Access Point \""); Serial.print(AP_SSID); Serial.println("\" iniciado");
  Serial.print("IP Address: "); Serial.println(WiFi.softAPIP());
  dnsServer.start(53, "*", AP_LOCAL_IP);
  server.on("/", handleAPRoot);
  server.on("/save", handleAPSave);
  server.onNotFound(handle_NotFound);
  server.begin();
  Serial.println("Servidor HTTP iniciado en modo AP");
}
void handleAPRoot() {
  String html = "<html><body><h1>Configuracion de Wi-Fi para ESP32 Servo</h1><p>Selecciona tu red y la contraseña:</p><form action=\"/save\" method=\"get\"><label for=\"ssid\">Red Wi-Fi (SSID):</label><br><select name=\"ssid\" id=\"ssid\">";
  int n = WiFi.scanNetworks();
  if (n == 0) { html += "<option value=\"\">No se encontraron redes</option>"; } else {
    for (int i = 0; i < n; ++i) { html += "<option value=\"" + WiFi.SSID(i) + "\">" + WiFi.SSID(i) + " (" + WiFi.RSSI(i) + " dBm)</option>"; delay(10); }
  }
  html += "</select><br><label for=\"pass\">Contrasena:</label><br><input type=\"password\" id=\"pass\" name=\"pass\"><br><br><input type=\"submit\" value=\"Conectar\"></form></body></html>";
  server.send(200, "text/html", html);
}
void handleAPSave() {
  if (server.hasArg("ssid") && server.hasArg("pass")) {
    STA_ssid = server.arg("ssid"); STA_password = server.arg("pass");
    preferences.putString("ssid", STA_ssid); preferences.putString("pass", STA_password);
    Serial.println("Guardando credenciales...");
    String html = "<html><body><h1>Configuracion Guardada!</h1><p>Intentando conectar a: " + STA_ssid + "</p><p>El ESP32 se reiniciara.</p></body></html>";
    server.send(200, "text/html", html);
    delay(2000); ESP.restart();
  } else { server.send(400, "text/plain", "Faltan parametros"); }
}

// --- Función para Conectar a WiFi en Modo STA ---
void connectToWiFi() {
  WiFi.mode(WIFI_STA);
  Serial.print("Conectando a la red: ");
  Serial.println(STA_ssid);
  WiFi.begin(STA_ssid.c_str(), STA_password.c_str());

  long startTime = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startTime < 30000) {
    delay(500); Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nConectado a la red Wi-Fi!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());

    // --- 2. CONFIGURACIÓN MDNS USANDO LA MAC COMPLETA (CORREGIDO) ---
    String mac = WiFi.macAddress();
    String hostname = mac;      // Copiamos la dirección MAC
    hostname.replace(":", "");  // ¡IMPORTANTE! Quitamos los dos puntos
    hostname.toLowerCase();     // Por convención, los hostnames van en minúscula

    if (MDNS.begin(hostname.c_str())) {
        Serial.println("MDNS responder iniciado");
        Serial.print("Hostname: http://");
        Serial.print(hostname);
        Serial.println(".local");
        MDNS.addService("http", "tcp", 80); // Anunciar el servicio web
    } else {
        Serial.println("Error iniciando MDNS!");
    }
    // --- FIN CONFIGURACIÓN MDNS ---

    wifiConfigured = true;
    server.on("/", handle_OnConnect);
    server.on("/servoOpen", handle_servoOpen);
    server.on("/servoClose", handle_servoClose);
    server.onNotFound(handle_NotFound);
    server.begin();
    Serial.println("Servidor HTTP iniciado en modo STA");

  } else {
    Serial.println("\nFallo la conexion a la red Wi-Fi.");
    STA_ssid = ""; STA_password = "";
    preferences.putString("ssid", ""); preferences.putString("pass", "");
    wifiConfigured = false;
    startAPMode();
  }
}

void setup() {
  Serial.begin(115200);
  servo.attach(servoPin);
  servo.write(0);
  preferences.begin("wifi-config", false);
  STA_ssid = preferences.getString("ssid", "");
  STA_password = preferences.getString("pass", "");

  if (STA_ssid != "" && STA_password != "") {
    connectToWiFi();
  } else {
    startAPMode();
  }
}

void loop() {
  if (!wifiConfigured) {
    dnsServer.processNextRequest();
  }
  server.handleClient();
}
