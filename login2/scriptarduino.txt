#include <WiFi.h>
#include <HTTPClient.h>
#include <ESP32Servo.h>
#include <Preferences.h>
#include <AsyncTCP.h>
#include <ESPAsyncWebServer.h>
#include <ArduinoJson.h> 
#include <map> 

// --- CONFIGURACIÓN ---
String serverUrl = "https://vecopo.ddns.net/vecopo/public/dispositivos/estado/";
const char* AP_SSID = "ESP32_Config_Servo";
const char* AP_PASSWORD = "password";
IPAddress AP_LOCAL_IP(192, 168, 4, 1);

// --- MULTIPLES SERVOS ---
// Definir los objetos Servo y los pines GPIO a los que estarán conectados
// Ajusta estos pines según tus conexiones físicas.
// Agrega más si tienes más de 3 servos.
Servo servoPin2;  // Objeto Servo para el pin 2
Servo servoPin4;  // Objeto Servo para el pin 4
Servo servoPin16; // Objeto Servo para el pin 16

// Para llevar el estado local de cada servo y evitar movimientos innecesarios
// Usaremos un mapa para mayor flexibilidad
std::map<int, String> servoLocalStates; // Requiere #include <map>

// Función para obtener el objeto Servo según el pin GPIO
Servo* getServoObject(int pin) {
    switch (pin) {
        case 2: return &servoPin2;
        case 4: return &servoPin4;
        case 16: return &servoPin16;
        // Agrega más casos si tienes más servos
        default: return nullptr; // Pin no reconocido
    }
}
// --- FIN MULTIPLES SERVOS ---

Preferences preferences;
bool isStaMode = false;
AsyncWebServer server(80);

void connectToWiFi();
void startAPMode();

void setup() {
    Serial.begin(115200);

    // Adjuntar cada servo a su pin GPIO
    // Asegúrate de que estos pines coincidan con tu hardware y con la tabla 'servos'
    servoPin2.attach(2);
    servoPin2.write(0); // CERRADO por defecto
    servoLocalStates[2] = "CERRADO";

    servoPin4.attach(4);
    servoPin4.write(0); // CERRADO por defecto
    servoLocalStates[4] = "CERRADO";

    servoPin16.attach(16);
    servoPin16.write(0); // CERRADO por defecto
    servoLocalStates[16] = "CERRADO";

    preferences.begin("wifi-creds", false);
    String STA_ssid = preferences.getString("ssid", "");
    String STA_password = preferences.getString("pass", "");
    if (STA_ssid.length() > 0) {
        isStaMode = true;
        connectToWiFi();
    } else {
        isStaMode = false;
        startAPMode();
    }
}

void loop() {
    if (isStaMode && WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        String mac = WiFi.macAddress();
        
        mac.replace(":", ""); 
        mac.toLowerCase();

        String fullUrl = serverUrl + mac;
        
        http.begin(fullUrl);
        int httpCode = http.GET();

        if (httpCode == 200) {
            String payload = http.getString();
            payload.trim();
            Serial.print("Payload recibido: '");
            Serial.print(payload);
            Serial.println("'");

            // --- PARSEAR EL NUEVO JSON CON ARRAY DE SERVOS ---
            StaticJsonDocument<512> doc; // Aumentar el tamaño del documento JSON para el array
                                        // Calcula el tamaño necesario con la calculadora de ArduinoJson
                                        // https://arduinojson.org/v6/assistant/

            DeserializationError error = deserializeJson(doc, payload);

            if (error) {
                Serial.print(F("Error al parsear JSON: "));
                Serial.println(error.f_str());
            } else {
                // Obtener el array de servos
                JsonArray servosArray = doc["servos"].as<JsonArray>();

                if (servosArray) { // Asegurarse de que el array "servos" existe
                    for (JsonObject servoObj : servosArray) {
                        int pin = servoObj["pin"];
                        String estado = servoObj["estado"].as<String>();
                        String modo = servoObj["modo"].as<String>();
                        estado.toUpperCase(); // Asegurarse que el estado sea "ABIERTO" o "CERRADO" en mayúsculas
                        modo.toUpperCase(); // Asegurarse que el modo sea "MANUAL" o "AUTOMATICO" en mayúsculas

                        Serial.print("Procesando Servo Pin: "); Serial.print(pin);
                        Serial.print(", Estado deseado: "); Serial.print(estado);
                        Serial.print(", Modo: "); Serial.println(modo);

                        Servo* currentServo = getServoObject(pin);

                        if (currentServo != nullptr) {
                            // Solo mover el servo si está en modo MANUAL o si el estado es diferente
                            if (modo == "MANUAL" || (servoLocalStates.count(pin) && servoLocalStates[pin] != estado)) {
                                if (estado == "ABIERTO") {
                                    currentServo->write(180);
                                    servoLocalStates[pin] = "ABIERTO";
                                    Serial.print("Servo Pin "); Serial.print(pin); Serial.println(" movido a ABIERTO");
                                } else if (estado == "CERRADO") {
                                    currentServo->write(0);
                                    servoLocalStates[pin] = "CERRADO";
                                    Serial.print("Servo Pin "); Serial.print(pin); Serial.println(" movido a CERRADO");
                                }
                            } else {
                                Serial.print("Servo Pin "); Serial.print(pin);
                                Serial.println(" no requiere movimiento (mismo estado o modo AUTOMATICO)");
                            }
                        } else {
                            Serial.print("Advertencia: Servo en pin "); Serial.print(pin); Serial.println(" no configurado en el código de la ESP.");
                        }
                    }
                } else {
                    Serial.println("Error: El payload no contiene un array 'servos' o está vacío.");
                }
            }
            // --- FIN PARSEO JSON ---
        } else {
            Serial.print("Error en la petición HTTP (código "); Serial.print(httpCode); Serial.print("): ");
            Serial.println(http.errorToString(httpCode).c_str());
        }
        http.end();
        delay(3000);
    } else if (isStaMode) {
        Serial.println("Wi-Fi desconectado, intentando reconectar...");
        connectToWiFi();
        delay(5000);
    } else {
        delay(100); 
    }
}

// Las funciones connectToWiFi() y startAPMode() son las mismas.
// No las incluyo aquí por brevedad, pero asegúrate de que estén en tu archivo.

void connectToWiFi() {
    String STA_ssid = preferences.getString("ssid", "");
    String STA_password = preferences.getString("pass", "");
    WiFi.mode(WIFI_STA);
    Serial.print("Conectando a: ");
    Serial.println(STA_ssid);
    WiFi.begin(STA_ssid.c_str(), STA_password.c_str());
    long startTime = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - startTime < 20000) {
        delay(500);
        Serial.print(".");
    }
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println("\n¡Conectado a la red Wi-Fi!");
        Serial.print("Dirección IP: "); Serial.println(WiFi.localIP());
    } else {
        Serial.println("\nFallo la conexión. Borrando credenciales y reiniciando en modo AP...");
        preferences.clear();
        delay(1000);
        ESP.restart();
    }
}

void startAPMode() {
    IPAddress AP_GATEWAY(192, 168, 4, 1);
    IPAddress AP_SUBNET(255, 255, 255, 0);
    Serial.println("Iniciando modo de configuración (AP)...");
    WiFi.softAP(AP_SSID, AP_PASSWORD);
    delay(500);
    WiFi.softAPConfig(AP_LOCAL_IP, AP_GATEWAY, AP_SUBNET);
    Serial.print("AP iniciado. Conéctate a '"); Serial.print(AP_SSID);
    Serial.print("' y ve a http://"); Serial.println(WiFi.softAPIP());

    server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
        String html = "<html><head><meta name='viewport' content='width=device-width, initial-scale=1'><style>body{font-family: Arial; text-align: center;} select, input{width: 80%; padding: 10px; margin: 10px 0;}</style></head><body><h1>Configurar Wi-Fi</h1><form action='/save' method='get'><select name='ssid'>";
        int n = WiFi.scanNetworks();
        for (int i = 0; i < n; ++i) { html += "<option value='" + WiFi.SSID(i) + "'>" + WiFi.SSID(i) + "</option>"; }
        html += "</select><br><input name='pass' type='password' placeholder='Contraseña'><br><br><input type='submit' value='Guardar y Conectar'></form></body></html>";
    
        request->send(200, "text/html", html);
    });
    server.on("/save", HTTP_GET, [](AsyncWebServerRequest *request){
        preferences.putString("ssid", request->getParam("ssid")->value());
        preferences.putString("pass", request->getParam("pass")->value());
        request->send(200, "text/plain", "Guardado. Reiniciando...");
        delay(1000);
        ESP.restart();
    });
    server.begin();
}
